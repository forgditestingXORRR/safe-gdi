-- ============================================================
-- CORE: SUPER FLING
-- ============================================================
workspace.ChildAdded:Connect(function(model)
    if model.Name == "GrabParts" then
        local grabPart = model:FindFirstChild("GrabPart")
        local weld = grabPart and grabPart:FindFirstChildOfClass("WeldConstraint")
        local partToImpulse = weld and weld.Part1

        if partToImpulse then
            local velocityObj = Instance.new("BodyVelocity", partToImpulse)
            velocityObj.MaxForce = Vector3.new(0, 0, 0)
            velocityObj.Velocity  = Vector3.zero

            model:GetPropertyChangedSignal("Parent"):Connect(function()
                if not model.Parent then
                    if superFlingEnabled then
                        local dir = Camera.CFrame.LookVector

                        if highFlingEnabled then
                            dir = (Camera.CFrame.LookVector + Vector3.new(0, 0.6, 0)).Unit
                        end

                        velocityObj.MaxForce = Vector3.new(math.huge, math.huge, math.huge)
                        velocityObj.Velocity  = dir * flingStrength

                        if spinFlingEnabled then
                            local bav = Instance.new("BodyAngularVelocity", partToImpulse)
                            bav.MaxTorque     = Vector3.new(math.huge, math.huge, math.huge)
                            bav.AngularVelocity = Vector3.new(0, 500, 0)
                            Debris:AddItem(bav, 0.5)
                        end

                        Debris:AddItem(velocityObj, 1)
                    else
                        velocityObj.MaxForce = Vector3.new(0, 0, 0)
                        Debris:AddItem(velocityObj, 0.1)
                    end
                end
            end)
        end
    end
end)

-- ============================================================
-- CORE: AUTO FLING (watches for GrabParts targeting local player)
-- ============================================================
workspace.ChildAdded:Connect(function(model)
    if not autoFlingEnabled then return end
    if model.Name ~= "GrabParts" then return end

    -- Check if this grab is targeting us
    local grabPart = model:WaitForChild("GrabPart", 1)
    if not grabPart then return end

    local weld = grabPart:FindFirstChildOfClass("WeldConstraint")
    if not weld then return end

    -- Part1 should be our HRP if we're the target
    local hrp = getHRP()
    if not hrp or weld.Part1 ~= hrp then return end

    -- We're being grabbed — immediately apply outward fling velocity
    task.spawn(function()
        local bv = Instance.new("BodyVelocity", hrp)
        bv.MaxForce = Vector3.new(math.huge, math.huge, math.huge)
        bv.Velocity  = Camera.CFrame.LookVector * flingStrength
        Debris:AddItem(bv, 0.4)

        -- Also fire struggle to break the weld server-side
        local CE = ReplicatedStorage:FindFirstChild("CharacterEvents")
        if CE then
            local struggle = CE:FindFirstChild("Struggle")
            if struggle then pcall(function() struggle:FireServer(LocalPlayer) end) end
        end

        Notify("Auto Fling", "Detected grab — auto-flung!")
    end)
end)

-- ============================================================
-- CORE: ANTI GRAB
-- ============================================================
local antiGrabConn = nil

local function startAntiGrabLoop()
    if antiGrabConn then antiGrabConn:Disconnect() antiGrabConn = nil end
    antiGrabConn = RunService.Heartbeat:Connect(function()
        if not antiGrabEnabled then
            antiGrabConn:Disconnect()
            antiGrabConn = nil
            return
        end
        local hrp = getHRP()
        local hum = getHum()
        local BeingHeld = LocalPlayer:FindFirstChild("IsHeld")
        if not hrp or not hum then return end

        if BeingHeld and BeingHeld.Value == true then
            hrp.AssemblyLinearVelocity = Vector3.zero
            hrp.Velocity = Vector3.zero
            hrp.RotVelocity = Vector3.zero

            local rayOrigin = hrp.Position
            local rayDir = Vector3.new(0, -100, 0)
            local rayParams = RaycastParams.new()
            rayParams.FilterDescendantsInstances = {LocalPlayer.Character}
            rayParams.FilterType = Enum.RaycastFilterType.Exclude
            local result = workspace:Raycast(rayOrigin, rayDir, rayParams)
            if result then
                hrp.CFrame = CFrame.new(result.Position + Vector3.new(0, 3, 0))
            end

            local CE = ReplicatedStorage:FindFirstChild("CharacterEvents")
            if CE then
                local StruggleEvent = CE:FindFirstChild("Struggle")
                if StruggleEvent then
                    pcall(function() StruggleEvent:FireServer(LocalPlayer) end)
                end
            end
        end
    end)
end

local function setupAntiGrab(char)
    if not char then return end
end

if LocalPlayer.Character then setupAntiGrab(LocalPlayer.Character) end
LocalPlayer.CharacterAdded:Connect(function(char)
    char:WaitForChild("Humanoid")
    setupAntiGrab(char)
    if antiGrabEnabled then startAntiGrabLoop() end
    local hum = char:FindFirstChildOfClass("Humanoid")
    if hum then hum.JumpPower = jumpPowerVal end
end)

-- ============================================================
-- CORE: ANTI EXPLOSION + ANTI ROCKET (fixed)
-- Sets BlastPressure = 0 immediately on any explosion added
-- Anti-Rocket additionally ground-locks for 0.8s after hit
-- ============================================================
workspace.DescendantAdded:Connect(function(v)
    if not v:IsA("Explosion") then return end

    -- Always zero blast pressure for both toggles
    if antiRocketEnabled or antiExplosionEnabled then
        v.BlastPressure = 0
        v.BlastRadius   = 0
    end

    -- Anti-Rocket: additionally lock the player to the ground
    if antiRocketEnabled then
        local hrp = getHRP()
        if not hrp then return end

        -- Disconnect any existing lock
        if rocketGroundConn then
            rocketGroundConn:Disconnect()
            rocketGroundConn = nil
        end

        local lockUntil = tick() + 0.8

        rocketGroundConn = RunService.Heartbeat:Connect(function()
            if tick() > lockUntil then
                rocketGroundConn:Disconnect()
                rocketGroundConn = nil
                return
            end

            local hrp2 = getHRP()
            if not hrp2 then
                rocketGroundConn:Disconnect()
                rocketGroundConn = nil
                return
            end

            -- Zero ALL velocity, not just upward
            hrp2.AssemblyLinearVelocity = Vector3.zero
            hrp2.Velocity    = Vector3.zero
            hrp2.RotVelocity = Vector3.zero

            -- Snap to ground via raycast
            local rayParams = RaycastParams.new()
            rayParams.FilterDescendantsInstances = {LocalPlayer.Character}
            rayParams.FilterType = Enum.RaycastFilterType.Exclude
            local result = workspace:Raycast(hrp2.Position, Vector3.new(0, -50, 0), rayParams)
            if result then
                hrp2.CFrame = CFrame.new(result.Position + Vector3.new(0, 3, 0))
            end
        end)
    end
end)

-- ============================================================
-- CORE: FLING AURA (proximity auto-fling anyone who gets close)
-- ============================================================
RunService.Heartbeat:Connect(function()
    if not flingAuraEnabled then return end
    local hrp = getHRP()
    if not hrp then return end

    for _, p in ipairs(Players:GetPlayers()) do
        if p ~= LocalPlayer and p.Character then
            local tHRP = p.Character:FindFirstChild("HumanoidRootPart")
            if tHRP then
                local dist = (hrp.Position - tHRP.Position).Magnitude
                if dist <= flingAuraRange then
                    -- Check no GrabParts exist already for this player
                    local bv = tHRP:FindFirstChild("AuraFlingBV")
                    if not bv then
                        local newBV = Instance.new("BodyVelocity", tHRP)
                        newBV.Name     = "AuraFlingBV"
                        newBV.MaxForce = Vector3.new(math.huge, math.huge, math.huge)
                        -- Direction: outward from our position
                        local dir = (tHRP.Position - hrp.Position).Unit
                        newBV.Velocity = dir * flingStrength
                        Debris:AddItem(newBV, 0.3)

                        if spinFlingEnabled then
                            local bav = Instance.new("BodyAngularVelocity", tHRP)
                            bav.MaxTorque     = Vector3.new(math.huge, math.huge, math.huge)
                            bav.AngularVelocity = Vector3.new(0, 500, 0)
                            Debris:AddItem(bav, 0.3)
                        end
                    end
                end
            end
        end
    end
end)

-- RENDER LOOP
-- ============================================================
RunService.RenderStepped:Connect(function(dt)
    local hrp = getHRP()
    local hum = getHum()

    if cframeSpeedEnabled and hrp and hum then
        if hum.MoveDirection.Magnitude > 0 then
            hrp.CFrame = hrp.CFrame + (hum.MoveDirection * cframeSpeedVal * 0.1)
        end
    end

    if flightEnabled and hrp then
        local v = Vector3.zero
        if UserInputService:IsKeyDown(Enum.KeyCode.W)         then v = v + Camera.CFrame.LookVector  end
        if UserInputService:IsKeyDown(Enum.KeyCode.S)         then v = v - Camera.CFrame.LookVector  end
        if UserInputService:IsKeyDown(Enum.KeyCode.A)         then v = v - Camera.CFrame.RightVector end
        if UserInputService:IsKeyDown(Enum.KeyCode.D)         then v = v + Camera.CFrame.RightVector end
        if UserInputService:IsKeyDown(Enum.KeyCode.Space)     then v = v + Vector3.new(0, 1, 0)      end
        if UserInputService:IsKeyDown(Enum.KeyCode.LeftShift) then v = v - Vector3.new(0, 1, 0)      end
        hrp.Velocity = Vector3.zero
        if v.Magnitude > 0 then
            hrp.CFrame = hrp.CFrame + (v.Unit * (flightSpeedVal / 10))
        end
    end

    if freezeSelfEnabled and hrp then
        hrp.Velocity               = Vector3.zero
        hrp.RotVelocity            = Vector3.zero
        hrp.AssemblyLinearVelocity = Vector3.zero
    end
end)

RunService.Stepped:Connect(function()
    local char = getChar()
    if noclipEnabled and char then
        for _, p in pairs(char:GetDescendants()) do
            if p:IsA("BasePart") then p.CanCollide = false end
        end
    end

    if antiVoidEnabled then
        local hrp = getHRP()
        if hrp and hrp.Position.Y < -100 then
            hrp.CFrame = CFrame.new(0, 50, 0)
            Notify("Anti-Void", "Saved from void!", 2)
        end
    end
end)

-- Grab All loop
task.spawn(function()
    while true do
        if grabAllEnabled then
            local hrp = getHRP()
            local hum = getHum()
            if hrp and hum then
                for _, p in ipairs(Players:GetPlayers()) do
                    if p ~= LocalPlayer and p.Character then
                        local tHRP = p.Character:FindFirstChild("HumanoidRootPart")
                        if tHRP then
                            local saved = hrp.CFrame
                            hum.PlatformStand = true
                            hrp.CFrame = tHRP.CFrame
                            task.wait(0.03)
                            hrp.CFrame = saved
                            hum.PlatformStand = false
                        end
                    end
                end
            end
        end
        task.wait(0.05)
    end
end)

MiscTab:CreateButton({
    Name     = "Copy My Position to Clipboard",
    Callback = function()
        local hrp = getHRP()
        if hrp then
            local p = hrp.Position
            setclipboard(string.format("Vector3.new(%.2f, %.2f, %.2f)", p.X, p.Y, p.Z))
            Notify("Position Copied", string.format("%.1f, %.1f, %.1f", p.X, p.Y, p.Z))
        end
    end
})

MiscTab:CreateButton({
    Name     = "Remove All Highlights",
    Callback = function()
        for _, p in ipairs(Players:GetPlayers()) do
            if p.Character then
                for _, obj in pairs(p.Character:GetDescendants()) do
                    if obj:IsA("Highlight") then obj:Destroy() end
                end
            end
        end
        Notify("Cleaned", "All highlights removed")
    end
})

MiscTab:CreateButton({
    Name     = "Kill All BodyMovers on Self",
    Callback = function()
        local hrp = getHRP()
        if hrp then
            for _, obj in pairs(hrp:GetChildren()) do
                if obj:IsA("BodyVelocity") or obj:IsA("BodyAngularVelocity")
                or obj:IsA("BodyForce") or obj:IsA("BodyPosition") then
                    obj:Destroy()
                end
            end
            Notify("Cleaned", "All body movers removed from HRP")
        end
    end
})
